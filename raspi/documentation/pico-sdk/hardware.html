하드웨어 API는 RP2040/RP2350 칩의 레지스터에 직접 대응하는 저수준 API로, 총 35개 모듈로 구성됩니다. 그중 실무에서 가장 자주 쓰이는 7개를 번역했습니다.

hardware_gpio

RP 시리즈 마이크로컨트롤러의 디지털 신호 제어를 위한 범용 입출력(GPIO) 핀을 관리합니다.

void gpio_init(uint gpio);                 // GPIO 핀 초기화
void gpio_set_dir(uint gpio, bool out);     // 입력/출력 방향 설정
void gpio_put(uint gpio, bool value);       // GPIO 핀에 디지털 값 쓰기
bool gpio_get(uint gpio);                   // GPIO 핀의 현재 상태 읽기
void gpio_set_function(uint gpio, enum gpio_function fn); // 대체 기능 설정
void gpio_pull_up(uint gpio);               // 내부 풀업 저항 활성화
void gpio_pull_down(uint gpio);             // 내부 풀다운 저항 활성화
void gpio_set_irq_enabled(uint gpio, uint32_t event_mask, bool enabled); // GPIO 인터럽트 설정
void gpio_set_irq_callback(gpio_irq_callback_t callback); // 인터럽트 콜백 등록

hardware_i2c

주변장치와의 동기식 직렬 데이터 교환을 위한 I2C(Inter-Integrated Circuit) 통신 인터페이스를 제공합니다.

void i2c_init(i2c_inst_t *i2c, uint baudrate);   // 지정한 속도로 I2C 컨트롤러 초기화
int i2c_write_blocking(i2c_inst_t *i2c, uint8_t addr, const uint8_t *src, size_t len, bool nostop); // 슬레이브에 데이터 전송
int i2c_read_blocking(i2c_inst_t *i2c, uint8_t addr, uint8_t *dst, size_t len, bool nostop); // 슬레이브로부터 데이터 수신
void i2c_set_baudrate(i2c_inst_t *i2c, uint baudrate); // 통신 속도 재설정
void i2c_set_slave_mode(i2c_inst_t *i2c, bool slave, uint8_t addr); // 마스터/슬레이브 모드 설정
bool i2c_get_write_available(i2c_inst_t *i2c);   // 송신 FIFO 여유 공간 확인

hardware_spi

외부 장치와의 고속 동기식 직렬 통신을 위한 SPI(Serial Peripheral Interface)를 구현합니다.

void spi_init(spi_inst_t *spi, uint baudrate);   // 지정한 클럭으로 SPI 컨트롤러 초기화
uint spi_write_blocking(spi_inst_t *spi, const uint8_t *src, size_t len); // 응답을 읽지 않고 데이터 전송
uint spi_read_blocking(spi_inst_t *spi, uint8_t repeat_tx_data, uint8_t *dst, size_t len); // 데이터 수신(더미 바이트 전송)
uint spi_write_read_blocking(spi_inst_t *spi, const uint8_t *src, uint8_t *dst, size_t len); // 동시 송수신
void spi_set_baudrate(spi_inst_t *spi, uint baudrate); // 초기화 이후 클럭 속도 변경
void spi_set_format(spi_inst_t *spi, uint data_bits, spi_cpol_t cpol, spi_cpha_t cpha, spi_order_t order); // 데이터 포맷/극성/위상/비트순서 설정

hardware_pwm

디지털 신호로 아날로그와 유사한 출력을 정밀하게 제어하는 PWM(Pulse Width Modulation) 신호를 생성합니다.

void pwm_init(uint slice_num, pwm_config *c, bool clk_en); // 설정값으로 PWM 슬라이스 초기화
void pwm_set_wrap(uint slice_num, uint16_t wrap);   // PWM 출력 주기(wrap 값) 설정
void pwm_set_chan_level(uint slice_num, uint chan, uint16_t level); // 채널별 듀티 사이클 설정
uint pwm_get_counter(uint slice_num);               // PWM 슬라이스의 현재 카운터 값 읽기
void pwm_set_enabled(uint slice_num, bool enabled); // 슬라이스의 PWM 생성 활성/비활성화
pwm_config pwm_get_default_config(void);            // 기본 PWM 설정 구조체 가져오기

hardware_adc

12비트 SAR(Successive Approximation Register) ADC로 아날로그 전압을 디지털 값으로 변환합니다.

void adc_init(void);                    // ADC 하드웨어 컨트롤러 초기화
void adc_gpio_init(uint gpio);          // GPIO 핀을 아날로그 입력용으로 준비
void adc_select_input(uint input);      // 샘플링할 ADC 입력 채널 선택
uint16_t adc_read(void);                // 단일 변환 수행 후 12비트 결과 반환
void adc_set_temp_sensor_enabled(bool enable); // 내장 온도 센서 켜기/끄기
void adc_run(bool run);                 // 연속(free-running) 변환 모드 활성화

hardware_timer

정밀한 타이밍과 인터럽트 스케줄링을 위한 하드웨어 타이머 저수준 접근을 제공합니다.

uint64_t timer_hw_read(void);            // 현재 64비트 타이머 카운터 값 읽기
void timer_hw_set_alarm(uint alarm_num, uint64_t target, timer_callback_t callback); // 지정 시각에 인터럽트 예약
void timer_hw_clear_alarm(uint alarm_num); // 예약된 타이머 인터럽트 취소
uint64_t timer_hw_get_alarm(uint alarm_num); // 현재 알람 목표값 조회
void timer_hw_pause(void);               // 타이머 카운팅 정지
void timer_hw_resume(void);              // 일시정지 상태에서 카운팅 재개

hardware_uart

UART(Universal Asynchronous Receiver/Transmitter)를 통한 비동기 직렬 통신을 관리합니다.

void uart_init(uart_inst_t *uart, uint baudrate); // 지정한 속도로 UART 초기화
void uart_putc(uart_inst_t *uart, char c);        // UART로 문자 1개 전송
int uart_getc(uart_inst_t *uart);                 // 문자 1개 수신(non-blocking)
void uart_puts(uart_inst_t *uart, const char *s); // null-종료 문자열 전송
void uart_set_baudrate(uart_inst_t *uart, uint baudrate); // 재초기화 없이 보드레이트 변경
bool uart_is_readable(uart_inst_t *uart);         // 수신 버퍼에 데이터가 있는지 확인
