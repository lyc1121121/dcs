고수준 API는 하드웨어 API 위에 편의 기능을 얹은 모듈들로, 실무에서 가장 자주 쓰이는 4개를 번역했습니다.

pico_stdlib

대부분의 실행 파일에서 공통으로 쓰이는 핵심 SDK 라이브러리들을 한데 모아 기본 기능과 유틸리티 메서드를 제공합니다.

void stdio_init_all(void);          // 표준 입출력 초기화
void sleep_ms(uint ms);             // 밀리초 단위 지연
uint64_t time_us_64(void);          // 현재 시각을 마이크로초로 반환
void gpio_init(uint gpio);          // GPIO 핀 초기화
void gpio_set_dir(uint gpio, bool out); // 입력/출력 방향 설정
bool gpio_get(uint gpio);           // GPIO 핀 상태 읽기
void gpio_put(uint gpio, bool value); // GPIO 핀에 값 쓰기

pico_multicore

보조 프로세서 코어(코어 1)에서 코드를 실행하고, FIFO·도어벨·락아웃을 통해 코어 간 통신을 관리합니다.

void multicore_launch_core1(void (*entry)(void)); // 코어 1에서 함수 실행
void multicore_fifo_push_blocking(uint32_t data); // 코어 간 FIFO에 값 push
uint32_t multicore_fifo_pop_blocking(void);       // FIFO에서 값 pop(blocking)
bool multicore_fifo_rvalid(void);                 // FIFO에 읽을 데이터가 있는지 확인
void multicore_reset_core1(void);                 // 코어 1 리셋

pico_time

타임스탬프, 슬립(대기), 시간 기반 콜백 스케줄링 기능을 제공합니다.

absolute_time_t get_absolute_time(void);          // 현재 절대 시각 가져오기
void sleep_until(absolute_time_t target);         // 지정 시각까지 대기
absolute_time_t make_timeout_time_ms(uint32_t ms);// 현재 시각 + N밀리초 계산
repeating_timer_t *add_repeating_timer_ms(int32_t ms, callback_t); // 반복 콜백 예약
bool cancel_repeating_timer(repeating_timer_t *timer); // 반복 타이머 취소

pico_sync

여러 코어와 인터럽트 컨텍스트에서 공유 자원 접근을 조율하는 동기화 기본요소를 제공합니다.

void mutex_init(mutex_t *mtx);                    // 뮤텍스 초기화
void mutex_enter_blocking(mutex_t *mtx);          // 뮤텍스 획득(blocking)
void mutex_exit(mutex_t *mtx);                    // 뮤텍스 해제
void sem_init(semaphore_t *sem, uint16_t initial_permit_count, uint16_t max_permits); // 세마포어 초기화
void sem_acquire_blocking(semaphore_t *sem);      // 세마포어 획득
void sem_release(semaphore_t *sem);               // 세마포어 반환
